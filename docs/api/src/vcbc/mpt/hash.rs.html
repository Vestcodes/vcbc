1//! Hashing and key utilities for MPT operations.
2//!
3//! Provides key conversion utilities and path manipulation functions
4//! for efficient trie operations.
5
6/// Key conversion utilities
7pub struct KeyUtils;
8
9impl KeyUtils {
10    /// Convert string key to nibbles (4-bit chunks)
11    pub fn string_to_nibbles(key: &str) -> Vec<u8> {
12        let mut nibbles = Vec::new();
13        for byte in key.as_bytes() {
14            nibbles.push(byte >> 4); // High 4 bits
15            nibbles.push(byte & 0x0F); // Low 4 bits
16        }
17        nibbles
18    }
19
20    /// Convert nibbles back to string
21    pub fn nibbles_to_string(nibbles: &[u8]) -> String {
22        let mut bytes = Vec::new();
23        for chunk in nibbles.chunks(2) {
24            if chunk.len() == 2 {
25                let byte = (chunk[0] << 4) | chunk[1];
26                bytes.push(byte);
27            }
28        }
29        String::from_utf8_lossy(&bytes).to_string()
30    }
31}
32
33/// Path manipulation utilities
34pub struct PathUtils;
35
36impl PathUtils {
37    /// Find the length of the common prefix between two paths
38    pub fn common_prefix_length(path1: &[u8], path2: &[u8]) -> usize {
39        path1
40            .iter()
41            .zip(path2.iter())
42            .take_while(|(a, b)| a == b)
43            .count()
44    }
45
46    /// Split a path at the given index
47    pub fn split_at(path: &[u8], index: usize) -> (&[u8], &[u8]) {
48        (&path[..index], &path[index..])
49    }
50}

52#[cfg(test)]
53mod tests {
54    use super::*;
55
56    #[test]
57    fn test_string_to_nibbles() {
58        let nibbles = KeyUtils::string_to_nibbles("a");
59        // 'a' is 97 in ASCII, which is 01100001 in binary
60        // High 4 bits: 0110 = 6, Low 4 bits: 0001 = 1
61        assert_eq!(nibbles, vec![6, 1]);
62
63        let nibbles = KeyUtils::string_to_nibbles("ab");
64        // 'a' -> [6, 1], 'b' -> [6, 2]
65        assert_eq!(nibbles, vec![6, 1, 6, 2]);
66    }
67
68    #[test]
69    fn test_nibbles_to_string() {
70        let nibbles = vec![6, 1];
71        let string = KeyUtils::nibbles_to_string(&nibbles);
72        assert_eq!(string, "a");
73
74        let nibbles = vec![6, 1, 6, 2];
75        let string = KeyUtils::nibbles_to_string(&nibbles);
76        assert_eq!(string, "ab");
77    }
78
79    #[test]
80    fn test_key_utils() {
81        // Test roundtrip conversion
82        let test_strings = vec!["", "a", "hello", "world", "🚀"];
83
84        for original in test_strings {
85            let nibbles = KeyUtils::string_to_nibbles(original);
86            let reconstructed = KeyUtils::nibbles_to_string(&nibbles);
87            assert_eq!(original, reconstructed);
88        }
89    }
90
91    #[test]
92    fn test_common_prefix_length() {
93        assert_eq!(PathUtils::common_prefix_length(&[1, 2, 3], &[1, 2, 3]), 3);
94        assert_eq!(PathUtils::common_prefix_length(&[1, 2, 3], &[1, 2, 4]), 2);
95        assert_eq!(PathUtils::common_prefix_length(&[1, 2, 3], &[4, 5, 6]), 0);
96        assert_eq!(PathUtils::common_prefix_length(&[], &[1, 2, 3]), 0);
97        assert_eq!(PathUtils::common_prefix_length(&[1, 2, 3], &[]), 0);
98    }
99
100    #[test]
101    fn test_split_at() {
102        let path = &[1, 2, 3, 4, 5];
103
104        let (left, right) = PathUtils::split_at(path, 0);
105        assert_eq!(left, &[] as &[u8]);
106        assert_eq!(right, &[1, 2, 3, 4, 5]);
107
108        let (left, right) = PathUtils::split_at(path, 2);
109        assert_eq!(left, &[1, 2]);
110        assert_eq!(right, &[3, 4, 5]);
111
112        let (left, right) = PathUtils::split_at(path, 5);
113        assert_eq!(left, &[1, 2, 3, 4, 5]);
114        assert_eq!(right, &[] as &[u8]);
115    }
